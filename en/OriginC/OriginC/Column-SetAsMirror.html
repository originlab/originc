Description

Set column as mirror

Syntax

BOOL	SetAsMirror( Column * pColSource, OSetMirror * pst = NULL )

Parameters

pColSource
[input] Source Column
OSetMirror
[input] Setting
struct OSetMirror
{
	DWORD	dwCntrl;//COL2MIRROR_*
	int	nSheetNameLabelRow;//Label Type to put Source Sheet, use with COL2MIRROR_SHEET_LABEL_ROW 
	int	nBookNameLabelRow;//Label Type to put Source Book, use with COL2MIRROR_BOOK_LABEL_ROW 
};

Return

Returns TRUE on successful exit and FALSE on failure.

Examples

void SetAsMirror_ex()
{	
	Worksheet		wks("[Book1]Sheet1");	
	Column			colSrc = wks.Columns(0);
	
	Worksheet	wksDest = Project.ActiveLayer();
	Column colDest = wksDest.Columns(0);	
	
	OSetMirror st;
	st.dwCntrl = COL2MIRROR_SHEET_LABEL_ROW;
	st.nSheetNameLabelRow = RCLT_COMMENT;

	colDest.SetAsMirror(&colSrc, &st);
}
