Improvement

keyword override

To extend or modify the abstract or virtual implementation of an inherited method, property. It can help to find out errors when base class’s prototype is modified but derived class is not. See more. 

class CTestBase
{
public:
	virtual void TestV()
	{
		printf("CTestBase::TestV\r\n");
	}
	virtual void TestV(int aa)
	{
		printf("CTestBase::TestV with int\r\n");
	}
};
class CTestObj : public CTestBase
{
public:
	CTestObj() {}
	void TestV() override;	
	void TestV(int aa, double dd) override //compile error, different from base class’s prototype
	{
	}
};

keyword __super

Call a base-class implementation for a function that you are overriding. See more. 

class base
{
public:
    virtual void fun(){ out_str("Base"); }
};
class A :public base
{
public:
    void fun(){ out_str("A");}
};
class B :public A
{
public:
    void fun()
    {
        __super::fun();// call A::fun()
        out_str("B");
    }
};
void main()
{
    B b;
    b.fun();
}

ffor() multi-threaded for loop

ffor is a multithreaded faster for statement. Inside ffor can call external VC function: 

//active matrix sheet should have 3 matrix object
// this will multiply the first two and put results on third one
void ffor_ex()
{
	MatrixPage mPage = Project.Pages();
	if(!mPage)
		return;
	MatrixLayer mlyr1 = mPage.Layers(0);
	Matrix matLayer1(mlyr1, 0);
	Matrix matLayer2(mlyr1, 1);
	Matrix matLayer3(mlyr1, 2);
	matrix mat1, mat2, mat3;
	mat1 = matLayer1;
	mat2 = matLayer2;
	mat3 = matLayer3;
	int nCols = mat1.GetNumCols();
	int nRows = mat1.GetNumRows();
	if(nRows != nCols)
		return;
	
	ffor(int nR; 0;  nRows)
	{
		ffor(int nC; 0; nRows)
		{
			double temp = 0;
			ffor(int k; 0; nRows)
			{
				double dmat = mat1[nR][k] * mat2[k][nC];
				AtomicAddDbl(&temp, dmat);
			}
			mat3[nR][nC] = temp;
		}
	}
	matLayer3 = mat3;
}

or call #pragma ffunc() function: 

void ffor_ffunc_ex(int n = 5)
{
 	Profiler  junk;

	v1.Data(5, n+5);

  	ffor(int ii; 0;  n)
  	{
   		v1[ii] = myAdd(v2[ii], v3[ii]);
   	}

#pragma ffunc(1)
int myAdd(int &n1, int &n2)
{
	myDuplicate(n1);
	myDuplicate(n2);
	return n1 + n2;
}
void myDuplicate(int &n)
{  
	n *= 2;
}
#pragma ffunc(0)

for() support iterating over range

//for ([int k, string v] : map) {}
//for (auto& [k, v] : map) {}
void for_map_ex()
{
	omap<int, string> m1;
	m1.Set(100, "Hello");
	m1.Set(200, "World");
	m1.Set(300, "test");
	for ([int k, string v] : m1)
	{
		printf("%d == %s\r\n", k, v);
	}
	for (auto& [key, val], index: m1)
	{
		printf("[%d] %d -> %s\r\n", index, key, val);
	}
}
